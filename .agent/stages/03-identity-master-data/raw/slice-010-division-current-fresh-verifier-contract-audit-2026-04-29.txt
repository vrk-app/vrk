   989	func normalizeOrganizationPropertyType(value string) (string, error) {
   990		normalized := strings.ToUpper(strings.TrimSpace(value))
   991		switch normalized {
   992		case "":
   993			return "", ErrPropertyTypeRequired
   994		case "LLC":
   995			normalized = "ООО"
   996		case "ОАО":
   997			normalized = "ПАО"
   998		case "ЗАО":
   999			normalized = "АО"
  1000		}
  1001		if _, ok := allowedOrganizationPropertyTypes[normalized]; !ok {
  1002			return "", ErrPropertyTypeInvalid
  1003		}
  1004		return normalized, nil
